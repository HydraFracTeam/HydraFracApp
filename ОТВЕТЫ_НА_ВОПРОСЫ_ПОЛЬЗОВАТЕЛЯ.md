#### 1. Архитектура QApplication для тестов

```python
@pytest.fixture(scope="module")
def qapp():
    """
    Создает единый экземпляр QApplication для всех GUI тестов.
    Scope="module" означает, что приложение создается один раз 
    для всего модуля тестов и переиспользуется.
    """
    if not QApplication.instance():
        # Создаем новое приложение, если его еще нет
        app = QApplication(sys.argv)
    else:
        # Используем существующее
        app = QApplication.instance()
    
    yield app  # Предоставляем приложение тестам
    
    # Cleanup после всех тестов (если нужно)
```

**Принцип работы:**
- QApplication - это глобальный объект Qt, может быть только один экземпляр
- `scope="module"` гарантирует переиспользование между тестами
- `yield` делает fixture генератором для cleanup'а

```python
@pytest.mark.performance
def test_button_response_time(self, app_with_data):
    """Измеряет время отклика кнопки"""
    app = app_with_data
    times = []
    
    # Выполняем несколько измерений для статистики
    for _ in range(20):
        # Засекаем время с высокой точностью
        start = time.perf_counter()
        
        # Вызываем метод кнопки
        app.on_plot_dimensionless_selected()
        
        # ВАЖНО: обрабатываем события Qt
        QApplication.instance().processEvents()
        
        # Вычисляем время в миллисекундах
        elapsed_ms = (time.perf_counter() - start) * 1000
        times.append(elapsed_ms)
    
    # Статистика
    avg_time = np.mean(times)
    max_time = np.max(times)
    
    # Проверяем критерии
    assert avg_time < 500, f"Среднее {avg_time:.2f}ms > 500ms"
    assert max_time < 1000, f"Максимальное {max_time:.2f}ms > 1000ms"
```

**Ключевые моменты:**
- `time.perf_counter()` - самый точный таймер (наносекунды)
- `processEvents()` - обязательно для обработки событий Qt
- Множественные измерения дают статистически значимый результат
- Проверяем и среднее, и максимальное время

```python
def test_memory_consumption(self, app_with_data):
    """Анализ потребления памяти"""
    import tracemalloc
    app = app_with_data
    
    # Запускаем трассировку памяти
    tracemalloc.start()
    
    # Базовый снимок
    snapshot1 = tracemalloc.take_snapshot()
    
    # Выполняем операции
    for _ in range(100):
        app.on_plot_dimensionless_selected()
        QApplication.instance().processEvents()
    
    # Конечный снимок
    snapshot2 = tracemalloc.take_snapshot()
    
    # Анализируем разницу
    top_stats = snapshot2.compare_to(snapshot1, 'lineno')
    total_memory_mb = sum(stat.size_diff for stat in top_stats) / 1024 / 1024
    
    tracemalloc.stop()
    
    # Проверяем
    assert total_memory_mb < 100, f"Память выросла на {total_memory_mb:.2f} МБ"
```

**Возможности tracemalloc:**
- Точное измерение в байтах
- Сравнение снимков памяти
- Определение источников утечек (файл:строка)

```python
def measure_response_time(button_method, iterations=20):

    for _ in range(iterations):

        QApplication.instance().processEvents()
        elapsed_ms = (time.perf_counter() - start) * 1000
        times.append(elapsed_ms)
    
    return {
        'mean': np.mean(times),
        'max': np.max(times),
        'min': np.min(times),
        'std': np.std(times)
```

```python
def test_failure_rate(iterations=1000):
    failures = 0
    exceptions = []
    
    for i in range(iterations):
        try:
            operation()
            QApplication.instance().processEvents()
        except Exception as e:
            failures += 1
            exceptions.append((i, str(e)))
    
    failure_rate = (failures / iterations) * 100
    
    print(f"Failure rate: {failure_rate:.2f}% ({failures}/{iterations})")
    if exceptions:
        print(f"Первые 5 исключений: {exceptions[:5]}")
    
    assert failure_rate < 1.0
```

```python
def test_memory_growth_is_linear():
    tracemalloc.start()
    
    # Измерения на разных точках
    measurements = []
    for target_iter in [10, 20, 50, 100, 200]:
        # Выполняем операции
        for _ in range(target_iter):
            operation()
        
        # Измеряем память
        snapshot = tracemalloc.take_snapshot()
        memory_mb = calculate_memory(snapshot)
        measurements.append((target_iter, memory_mb))
    
    # Линейная регрессия: memory = a * iterations + b
    iterations = np.array([m[0] for m in measurements])
    memory = np.array([m[1] for m in measurements])
    
    linear_fit = np.polyfit(iterations, memory, 1)  # степень 1
    linear_r2 = calculate_r_squared(linear_fit, iterations, memory)
    
    # Квадратичная регрессия: memory = a * iterations² + b * iterations + c
    quad_fit = np.polyfit(iterations, memory, 2)  # степень 2
    quad_r2 = calculate_r_squared(quad_fit, iterations, memory)
    
    # Проверки
    assert linear_r2 > 0.8, "Линейная модель недостаточно точна"
    assert (quad_r2 - linear_r2) < 0.1, "Рост сильно нелинейный"
```

### Критерии

- **R² > 0.8** для линейной модели - хорошее соответствие
- **Улучшение от квадратичной < 0.1** - рост близок к линейному, не квадратичному
- **Коэффициент роста** - разумный (не слишком большой)

### Интерпретация R²

| R² | Интерпретация | Статус |
|----|---------------|--------|
| > 0.9 | Отличный линейный рост | ✅✅ |

```python
def test_interpolation_accuracy():
    # Чистый синтетический сигнал
    time = pd.Series(np.linspace(0, 10, 100))
    true_values = pd.Series(np.sin(time) * 50 + 100)
    
    # Добавляем пропуски (12%)
    values_with_gaps = true_values.copy()
    gap_indices = [10, 11, 12, 30, 31, 32, ...]
    values_with_gaps.iloc[gap_indices] = np.nan
    
    # Интерполируем
    interpolated = apply_ml_interpolation(time, values_with_gaps, 'random_forest')
    
    # Вычисляем точность на пропущенных значениях
    true_gaps = true_values.iloc[gap_indices]
    interpolated_gaps = interpolated.iloc[gap_indices]
    
    relative_errors = |true_gaps - interpolated_gaps| / |true_gaps|
    accuracy = (1 - mean(relative_errors)) * 100%
    
    assert accuracy > 95.0
```

#### Фильтрация

```python
def test_filtering_accuracy():
```

### Методология

```python
def test_long_running_performance_degradation():
    # Измеряем начальную производительность
    initial_times = []
    for _ in range(50):
        start = time.perf_counter()
        operation()
        QApplication.instance().processEvents()
        initial_times.append(time.perf_counter() - start)
    
    initial_avg = np.mean(initial_times)
    
    # Долгая работа (1000 операций)
    print("Выполнение 1000 операций...")
    for i in range(1000):
        operation()
        QApplication.instance().processEvents()
        
        if i % 200 == 0:
            print(f"Прогресс: {i}/1000")
    
    # Измеряем конечную производительность
    final_times = []
    for _ in range(50):
        start = time.perf_counter()
        operation()
        QApplication.instance().processEvents()
        final_times.append(time.perf_counter() - start)
    
    final_avg = np.mean(final_times)
    
    # Вычисляем деградацию
    degradation = ((final_avg - initial_avg) / initial_avg) * 100
    
    print(f"Начальная: {initial_avg*1000:.2f} мс")
    print(f"Конечная: {final_avg*1000:.2f} мс")
    print(f"Деградация: {degradation:.2f}%")
    
    assert degradation < 15.0
```

### Формула деградации

```
Деградация = ((Конечная_производительность - Начальная) / Начальная) × 100%
```

### Интерпретация

| Деградация | Интерпретация | Статус |
|-----------|---------------|--------|
| < 5% | Отличная стабильность | ✅✅ |
| 5-10% | Хорошая стабильность | ✅ |
| 10-15% | Приемлемая стабильность | ⚠️ |
| > 15% | Проблемы со стабильностью | ❌ |
